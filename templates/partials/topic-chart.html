{#- Timeline of a section's posts by topic, computed from their front matter (date, extra.topic, extra.ai-assisted) and Zola's word count, so new posts show up on the next build. Topics and their order live in data/topics.yaml; the chart is skipped when no page of the section has a topic. -#} {%- set topics = load_data(path="data/topics.yaml") -%} {%- set_global count = 0 -%}{%- set_global first = 0 -%}{%- set_global last = 0 -%}{%- set_global longest = 1 -%} {%- for item in section.pages -%}{%- if item.extra.topic -%} {%- set time = item.date | date(format="%s") | int -%} {%- if count == 0 or time < first %}{% set_global first = time %}{% endif -%} {%- if time > last %}{% set_global last = time %}{% endif -%} {%- if item.word_count > longest %}{% set_global longest = item.word_count %}{% endif -%} {%- set_global count = count + 1 -%} {%- endif -%}{%- endfor -%} {%- if count > 0 -%} {#- The axis starts on the first of the first post's month and ends three days after the last post, so the newest dot is not pinned to the edge. -#} {%- set start_year = first | date(format="%Y") | int -%} {%- set start_month = first | date(format="%m") | int -%} {%- set start = first | date(format="%Y-%m-01") | date(format="%s") | int -%} {%- set end = last + 259200 -%} {%- set span = end - start -%} {%- set end_year = last | date(format="%Y") | int -%} {%- set end_month = last | date(format="%m") | int -%} {%- set months = end_year * 12 + end_month - start_year * 12 - start_month + 1 -%} {%- endif -%}
